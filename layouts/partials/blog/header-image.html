{{- /* Returns the URL of a post's generated header image, or "" if it has no `header` block. */ -}} {{- /* Templates are configured in data/header_templates.yaml; usage is in contributing-blog-posts.md. */ -}} {{- /* Call with the page for the on-page image, or with (dict "page" . "social" true) for og:image. */ -}} {{- $page := . -}} {{- $social := false -}} {{- if reflect.IsMap . -}} {{- $page = .page -}} {{- $social = .social -}} {{- end -}} {{- $url := "" -}} {{- with $page.Params.header -}} {{- $header := . -}} {{- $tpl := index hugo.Data.header_templates .template -}} {{- if not $tpl -}} {{- errorf "%s: unknown header template %q (see data/header_templates.yaml)" $page.File.Path .template -}} {{- end -}} {{- $bg := resources.Get (printf "header-templates/%s" $tpl.background) -}} {{- if not $bg -}} {{- errorf "header template %q: background assets/header-templates/%s not found" .template $tpl.background -}} {{- end -}} {{- /* A post can override the template's text options with header.style.title / .subtitle / .caption. */ -}} {{- $styles := dict -}} {{- range $field := slice "title" "subtitle" "caption" -}} {{- $override := index ($header.style | default dict) $field | default dict -}} {{- range $key, $_ := $override -}} {{- if not (isset (index $tpl $field | default dict) $key) -}} {{- errorf "%s: unknown header style option %s.%s (see data/header_templates.yaml)" $page.File.Path $field $key -}} {{- end -}} {{- end -}} {{- with index $tpl $field -}} {{- $styles = merge $styles (dict $field (merge . $override)) -}} {{- end -}} {{- end -}} {{- $filters := slice -}} {{- range $field := slice "title" "subtitle" -}} {{- with index $header $field -}} {{- $style := index $styles $field -}} {{- $opts := merge $style (dict "font" (resources.Get $style.font)) -}} {{- $filters = $filters | append (images.Text . $opts) -}} {{- end -}} {{- end -}} {{- /* Pictures from static/ (header.images), each with an optional caption centered below it. */ -}} {{- $picHashes := slice -}} {{- range $item := $header.images -}} {{- $file := path.Join "static" .src -}} {{- if not (fileExists $file) -}} {{- errorf "%s: header image %s not found in static/" $page.File.Path .src -}} {{- end -}} {{- $content := os.ReadFile $file -}} {{- $picHashes = $picHashes | append (md5 $content) -}} {{- $pic := (resources.FromString (path.Join "header-images" .src) $content).Fill (printf "%dx%d Lanczos" (int .width) (int .height)) -}} {{- $filters = $filters | append (images.Overlay $pic .x .y) -}} {{- with .caption -}} {{- $style := index $styles "caption" -}} {{- if not $style -}} {{- errorf "%s: header template %q has no caption style for image captions" $page.File.Path $header.template -}} {{- end -}} {{- $opts := dict "font" (resources.Get $style.font) "size" $style.size "color" $style.color "linespacing" $style.linespacing "x" (add $item.x (div $item.width 2)) "y" (add $item.y $item.height $style.margin) "alignx" "center" "aligny" "top" -}} {{- $filters = $filters | append (images.Text . $opts) -}} {{- end -}} {{- end -}} {{- /* Defaults are documented in data/header_templates.yaml. */ -}} {{- $format := $tpl.format | default "resize 1700x webp q85 Lanczos text" -}} {{- if $social -}} {{- $format = $tpl.social_format | default "resize 1200x jpg q85 Lanczos" -}} {{- end -}} {{- $img := ($bg | images.Filter $filters).Process $format -}} {{- /* Readable file name; the hash changes with the text and pictures so social media caches refresh. */ -}} {{- $hash := substr (md5 (print (jsonify $header) (delimit $picHashes ""))) 0 8 -}} {{- $name := printf "blog-images/generated/%s-%s%s" $page.File.ContentBaseName $hash $img.MediaType.FirstSuffix.FullSuffix -}} {{- $url = ($img | resources.Copy $name).RelPermalink -}} {{- end -}} {{- return $url -}}
